  93: def _solve_and_fill_captcha(page: Page, dump_dir: str | None = None) -> None:
  94:     """Wait for the real 120x50 SVG, OCR with retries, auto-refresh on failure."""
  95:     sel = CAPTCHA_SVG_SEL
  96:     img = page.locator(sel).first
  97:     try:
  98:         img.wait_for(state="visible", timeout=5000)  # ensure rendered
  99:     except Exception:
 100:         log("INFO", "captcha", "captcha svg not visible"); return
 101: 
 102:     # retry up to 3 times with '重新取得'
 103:     for attempt in range(1, 4):
 104:         png = img.screenshot(type="png")
 105: 
 106:         if dump_dir:
 107:             Path(dump_dir).mkdir(parents=True, exist_ok=True)
 108:             ts = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
 109:             Path(dump_dir, f"captcha_{ts}_try{attempt}.png").write_bytes(png)
 110: 
 111:         code = read_4digit_from_png_bytes(png)
 112:         if len(code) == 4:
 113:             page.locator(CAPTCHA_INPUT_SEL).fill(code)
 114:             log("INFO", "captcha", f"OCR ok on try {attempt}")
 115:             return
 116: 
 117:         # refresh and wait for svg to change
 118:         log("WARN", "captcha", f"OCR failed on try {attempt}, refreshing")
 119:         refresh = page.get_by_role("button", name="重新取得").first
 120:         if refresh.count() == 0:
 121:             # fallback: just small delay then retry
 122:             page.wait_for_timeout(600)
 123:         else:
 124:             old_html = img.evaluate("el => el.outerHTML")
 125:             refresh.click()
 126:             # wait until outerHTML changes (new captcha)
 127:             page.wait_for_function(
 128:                 "(selector, oldHtml) => { const el = document.querySelector(selector); return el && el.outerHTML !== oldHtml; }",
 129:                 (sel, old_html),
 130:                 timeout=5000
 131:             )
 132:             img = page.locator(sel).first  # rebind locator
 133: 
 134:     raise RuntimeError("captcha OCR failed after 3 attempts")

 136: def _fill_captcha_once(page: Page, attempt: int, dump_dir: str | None = None) -> bool:
 137:     """Try a single OCR read and fill. Return True if 4-digit code was filled."""
 138:     sel = CAPTCHA_SVG_SEL
 139:     img = page.locator(sel).first
 140:     try:
 141:         img.wait_for(state="visible", timeout=CAPTCHA_VISIBLE_TIMEOUT_MS)
 142:     except Exception:
 143:         log("INFO", "captcha", "captcha svg not visible")
 144:         return False
 145: 
 146:     png = img.screenshot(type="png")
 147:     if dump_dir:
 148:         try:
 149:             Path(dump_dir).mkdir(parents=True, exist_ok=True)
 150:             ts = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
 151:             Path(dump_dir, f"captcha_{ts}_try{attempt}.png").write_bytes(png)
 152:         except Exception:
 153:             pass
 154: 
 155:     code = read_4digit_from_png_bytes(png)
 156:     if len(code) == 4:
 157:         page.locator(CAPTCHA_INPUT_SEL).fill(code)
 158:         log("INFO", "captcha", f"OCR ok on try {attempt}")
 159:         return True
 160:     log("WARN", "captcha", f"OCR did not yield 4 digits on try {attempt}")
 161:     return False

 163: def _did_login_succeed(page: Page, *, timeout_ms: int = 2000) -> bool:
 164:     """Return True if we appear to have left the login page.
 165: 
 166:     Primary check: the login button disappears. Secondary signals: app shell attached or URL changed.
 167:     """
 168:     # Prefer the explicit signal: login button detached/hidden
 169:     try:
 170:         # Wait a bit for the button to disappear after click
 171:         btn = _login_button_locator(page)
 172:         if btn.count() > 0:
 173:             try:
 174:                 btn.wait_for(state="detached", timeout=timeout_ms)
 175:                 return True
 176:             except Exception:
 177:                 # If still attached, check visibility
 178:                 try:
 179:                     if not btn.is_visible():
 180:                         return True
 181:                 except Exception:
 182:                     pass
 183:     except Exception:
 184:         # Locator resolution failed; treat as likely navigated
 185:         return True
 186: 
 187:     # Fallbacks: shell attached or url matches
 188:     try:
 189:         page.wait_for_selector(APP_SHELL_SEL, state="attached", timeout=800)
 190:         return True
 191:     except Exception:
 192:         pass
 193:     try:
 194:         page.wait_for_url(CASE_URL_GLOB, timeout=800)
 195:         return True
 196:     except Exception:
 197:         pass
 198:     return False
